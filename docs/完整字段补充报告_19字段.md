protected void onCreate() {
    Date now = new Date();
    
    if (createTime == null) createTime = now;
    if (supCreateTime == null) supCreateTime = now;
    if (supUpdateTime == null) supUpdateTime = now;
    if (status == null) status = 0;
    
    // 默认值兜底
    if (shift == null || shift.trim().isEmpty()) shift = "0";
    if (machineNo == null || machineNo.trim().isEmpty()) machineNo = "NA";
}
